cmake_minimum_required(VERSION 3.10)
project(dog_description)
find_package(ament_cmake REQUIRED)
find_package(xacro REQUIRED)
find_package(robot_state_publisher REQUIRED)
find_package(rclpy REQUIRED)
find_package(gazebo_msgs REQUIRED)
find_package(gazebo_ros REQUIRED)
find_package(gazebo_dev REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
add_library(dog_set_model_configuration SHARED src/set_model_configuration_plugin.cpp)
ament_target_dependencies(dog_set_model_configuration gazebo_ros gazebo_dev gazebo_msgs rclcpp std_srvs)
install(TARGETS dog_set_model_configuration LIBRARY DESTINATION lib)
install(PROGRAMS scripts/set_initial_joint_positions.py DESTINATION lib/${PROJECT_NAME})
install(FILES package.xml DESTINATION share/${PROJECT_NAME})
install(DIRECTORY xacro config urdf launch meshes DESTINATION share/${PROJECT_NAME})
ament_package()
